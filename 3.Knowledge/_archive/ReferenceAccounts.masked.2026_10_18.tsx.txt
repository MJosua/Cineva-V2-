import { Link, useLocation } from 'react-router-dom';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import accounts from './accounts.json';

const displayNumber = (value: number | null) => value === null ? 'Belum tersedia' : value.toLocaleString('id-ID');
/** Read-only source observations: never writes jobs, payment ledgers or identity storage. */
export function ReferenceAccounts({ basePath }: { basePath: string }) {
  const location = useLocation();
  const requested = new URLSearchParams(location.search).get('account');
  const selected = accounts.find(account => account.id === requested) ?? accounts[0];
  const insight = selected.insight;
  return <section className="space-y-5" data-testid="reference-accounts">
    <div className="rounded-md border border-border bg-muted/40 p-4 text-sm">
      <Badge variant="secondary">Referensi Excel · hanya baca</Badge>
      <p className="mt-2">Tiga akun dipilih acak dari workbook. Nama, akun sosial, kontak, alamat dan URL pribadi disamarkan. Ini snapshot sumber, bukan login pengguna atau transaksi terverifikasi.</p>
      <p className="mt-1">Kosong berarti belum tersedia. Referensi link atau tulisan DONE tidak otomatis membuktikan QC, posting, atau pembayaran selesai.</p>
    </div>
    <nav className="flex flex-wrap gap-2" aria-label="Pilih akun referensi">
      {accounts.map(account => <Link key={account.id} aria-current={selected.id === account.id ? 'page' : undefined} className={`rounded-md border px-4 py-2 text-sm font-medium ${selected.id === account.id ? 'border-primary bg-primary text-primary-foreground' : 'border-border bg-card text-foreground'}`} to={`${basePath}/creator/reference?account=${account.id}`}>{account.alias}</Link>)}
    </nav>
    <Card><CardHeader><CardTitle>Hai, {selected.alias}.</CardTitle></CardHeader><CardContent className="space-y-2 text-sm">
      <p>Campaign: <strong>{selected.sheet}</strong></p>
      <p>Produk terpilih: {selected.product}</p>
      <p className="text-muted-foreground">Identitas asli dan detail pribadi tidak dimuat ke halaman ini.</p>
    </CardContent></Card>
    <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
      {[['Fee cash', 'Belum tersedia'], ['Cash sudah diterima', 'Belum terverifikasi'], ['Views sumber', displayNumber(insight?.views ?? null)], ['ER berdasarkan views', insight?.er === null || insight?.er === undefined ? 'Belum tersedia' : `${insight.er.toLocaleString('id-ID', { maximumFractionDigits: 2 })}%`]].map(([label, value]) => <Card key={label}><CardContent className="space-y-2 p-5"><p className="text-sm text-muted-foreground">{label}</p><p className="text-xl font-semibold">{value}</p></CardContent></Card>)}
    </div>
    <Card><CardHeader><CardTitle>Akun sosial</CardTitle></CardHeader><CardContent className="space-y-3">
      {selected.social.map(profile => <p key={profile.platform} className="text-sm">{profile.platform} · identitas disamarkan · Followers: {displayNumber(profile.followers)}</p>)}
      <p className="text-xs text-muted-foreground">Followers hanya memakai nilai numerik pada baris sumber. Kategori teks tidak diubah menjadi nol.</p>
    </CardContent></Card>
    <Card><CardHeader><CardTitle>Kolaborasi dan progress sumber</CardTitle></CardHeader><CardContent className="space-y-3">
      {selected.tasks.map((task, index) => <div key={`${task.cell}-${index}`} className="flex flex-wrap items-start justify-between gap-2 rounded-md border border-border p-3 text-sm"><div><p className="font-medium">{task.label}</p><p className="mt-1">{task.observation}</p></div><Badge variant="outline">{task.cell}</Badge></div>)}
      <p className="text-xs text-muted-foreground">Tidak mengubah STATUS PAYMENT menjadi Paid dan tidak menebak paid_amount, fee per talent, deadline atau tanggal posting dari data yang ambigu.</p>
    </CardContent></Card>
    <Card><CardHeader><CardTitle>Insight</CardTitle></CardHeader><CardContent className="space-y-3">
      {insight ? <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">{(['likes', 'comments', 'shares', 'saves', 'reposts', 'views'] as const).map(key => <div key={key}><p className="text-sm capitalize text-muted-foreground">{key}</p><p className="font-semibold">{displayNumber(insight[key])}</p></div>)}</div> : <p className="text-sm">Belum ada insight yang cocok secara unik untuk akun ini.</p>}
      <p className="text-xs text-muted-foreground">ER = (likes + comments + shares + saves + reposts) / views × 100. Nilai kosong atau views 0 menghasilkan belum tersedia. Tidak membuat kategori performa otomatis.</p>
    </CardContent></Card>
    <details className="rounded-md border border-border p-4 text-sm"><summary className="cursor-pointer font-medium">Asal data dan batas preview</summary><div className="mt-3 space-y-2 break-words"><p>{selected.sourceFile} · {selected.sheet} · baris roster {selected.sourceRow}{insight ? ` · insight baris ${insight.sourceRow}` : ''}</p><p>{selected.matchNote}</p><p>Data dibaca dari cache Excel, tanpa recalculation Excel. Sumber asli tidak diubah. Snapshot tidak masuk localStorage, data dummy, atau database.</p></div></details>
  </section>;
}
