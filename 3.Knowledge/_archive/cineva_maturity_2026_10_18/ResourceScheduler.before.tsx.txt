import { useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Checkbox } from '@/components/ui/checkbox';
import { validateDraft } from '../cineva-shared/model';
import { useWorkspace } from '../cineva-shared/WorkspaceContext';
export function ResourceScheduler() {
    const { data, mutate, busy, error } = useWorkspace(), [jobId, setJobId] = useState(''), [date, setDate] = useState('2026-10-19T10:00'), [ids, setIds] = useState<string[]>([]);
    if (!data)
        return null;
    return <Card><CardHeader><CardTitle>Alokasi crew & gear</CardTitle></CardHeader><CardContent><form className="space-y-4" onSubmit={async (e) => { e.preventDefault(); await mutate(current => { const job = current.jobs.find(j => j.id === jobId); if (!job)
        throw new Error('Pilih project terlebih dahulu.'); const next = { ...job, scheduledAt: date, resourceIds: ids }; validateDraft(next, current, jobId); return { ...current, jobs: current.jobs.map(row => row.id === jobId ? next : row) }; }); }}><label className="block space-y-2 text-sm">Project<select aria-label="Project untuk alokasi" required value={jobId} onChange={e => { setJobId(e.target.value); const job = data.jobs.find(j => j.id === e.target.value); if (job) {
        setDate(job.scheduledAt);
        setIds(job.resourceIds);
    } }} className="h-10 w-full rounded-md border border-input bg-background px-3"><option value="">Pilih project...</option>{data.jobs.filter(j => !j.archived).map(job => <option key={job.id} value={job.id}>{job.title}</option>)}</select></label><label className="block space-y-2 text-sm">Mulai (WIB)<Input aria-label="Mulai alokasi" type="datetime-local" required value={date} onChange={e => setDate(e.target.value)}/></label><div className="grid gap-3 sm:grid-cols-2">{data.resources.map(resource => <label key={resource.id} className="flex items-center gap-2 text-sm"><Checkbox checked={ids.includes(resource.id)} onCheckedChange={checked => setIds(old => checked ? [...old, resource.id] : old.filter(id => id !== resource.id))}/>{resource.name}</label>)}</div><p className="text-xs text-muted-foreground">Durasi mengikuti project. Resource yang overlap dengan project lain akan ditolak.</p>{error && <p role="alert" className="text-destructive">{error}</p>}<Button disabled={busy}>Simpan alokasi</Button></form></CardContent></Card>;
}
